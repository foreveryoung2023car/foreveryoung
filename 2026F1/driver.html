/** * 鈴鹿賽車接駁 - 司機簽到系統 API v3 * 配合 writeAssignment_v7 的新統整表結構 * * 統整Sheet 欄位（A~V，共22欄）： * A 區段（去程/回程） * B 日期 * C 出發時間 * D 預計抵達 * E 車號 * F 車牌 * G 車款 * H 客座 * I 上/下車地點 * J 回程時段 * K 訂單編號 * L 姓名 * M 當日人數 * N 車上總人數 * O 去程確認 * P 去程確認時間 * Q 回程確認 * R 回程確認時間 * S 去程已出發 * T 去程已送達 * U 回程已出發 * V 回程已送達 */ const SHEET_NAME = "統整"; const COL = { SEGMENT: 0, // A 區段（去程/回程） DATE: 1, // B 日期 DEP_TIME: 2, // C 出發時間 ARR_TIME: 3, // D 預計抵達 VID: 4, // E 車號 PLATE: 5, // F 車牌 VTYPE: 6, // G 車款 CAP: 7, // H 客座 DEP_PT: 8, // I 上/下車地點 RET_TIME: 9, // J 回程時段 ORDER_NO: 10, // K 訂單編號 NAME: 11, // L 姓名 PAX: 12, // M 當日人數 TOTAL: 13, // N 車上總人數 OUT_STATUS: 14, // O 去程確認 OUT_TIME: 15, // P 去程確認時間 RET_STATUS: 16, // Q 回程確認 RET_TIME2: 17, // R 回程確認時間 OUT_DEPART: 18, // S 去程已出發 OUT_ARRIVE: 19, // T 去程已送達 RET_DEPART: 20, // U 回程已出發 RET_ARRIVE: 21, // V 回程已送達 }; // ============================================= // GET // ============================================= function doGet(e) { const action = e.parameter.action || ""; const car = e.parameter.car || ""; const date = e.parameter.date || ""; let result; try { if (action === "getPassengers") result = getPassengers(car, date); else if (action === "getDashboard") result = getDashboard(date); else if (action === "findCar") result = findCar(e.parameter.num || "", date); else if (action === "findPassenger") result = findPassenger(e.parameter.phone || "", e.parameter.name || ""); else if (action === "debug") result = debugSheet(); else if (action === "stats") result = getStats(); else result = { error: "未知 action" }; } catch(err) { result = { error: err.message }; } return jsonOut(result); } // ============================================= // POST // ============================================= function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch(err) { return jsonOut({ error: "JSON 格式錯誤" }); } const { action, orderNo, date, trip, car, time } = body; let result; try { if (action === "confirm") result = updateStatus(orderNo, date, trip, "✅"); else if (action === "absent") result = updateStatus(orderNo, date, trip, "❌"); else if (action === "reset") result = updateStatus(orderNo, date, trip, ""); else if (action === "out_depart") result = updateCarCol(car, date, "去程", COL.OUT_DEPART, time||now()); else if (action === "out_arrive") result = updateCarCol(car, date, "去程", COL.OUT_ARRIVE, time||now()); else if (action === "ret_depart") result = updateCarCol(car, date, "回程", COL.RET_DEPART, time||now()); else if (action === "ret_arrive") result = updateCarCol(car, date, "回程", COL.RET_ARRIVE, time||now()); else if (action === "out_depart_clr") result = updateCarCol(car, date, "去程", COL.OUT_DEPART, ""); else if (action === "out_arrive_clr") result = updateCarCol(car, date, "去程", COL.OUT_ARRIVE, ""); else if (action === "ret_depart_clr") result = updateCarCol(car, date, "回程", COL.RET_DEPART, ""); else if (action === "ret_arrive_clr") result = updateCarCol(car, date, "回程", COL.RET_ARRIVE, ""); else result = { error: "未知 action" }; } catch(err) { result = { error: err.message }; } return jsonOut(result); } function jsonOut(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ============================================= // 取得乘客名單（司機用） // car: 車牌, date: "0329", trip: "outbound"/"return" // ============================================= function getPassengers(car, date) { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const dateStr = fmtDate(date); // 去程資訊 let depInfo = null; const depPassengers = []; // 回程資訊 let retInfo = null; const retPassengers = []; let lastSegment = ""; let lastOutArrive = ""; let lastRetDepart = ""; let lastRetArrive = ""; for (let i = 1; i < data.length; i++) { const row = data[i]; const rowSeg = String(row[COL.SEGMENT] || "").trim(); const rowDate = String(row[COL.DATE] || "").trim(); const rowPlate = String(row[COL.PLATE] || "").trim(); const rowOrder = String(row[COL.ORDER_NO]|| "").trim(); if (rowSeg) lastSegment = rowSeg; if (rowDate) lastDate = rowDate; if (!lastDate.includes(dateStr)) continue; if (rowPlate && rowPlate !== "？") { lastPlate = rowPlate; lastVid = String(row[COL.VID]).trim(); lastVtype = String(row[COL.VTYPE]).trim(); lastDepPt = String(row[COL.DEP_PT]).trim(); lastDepTime = fmtTime(row[COL.DEP_TIME]); lastRetTime = fmtTime(row[COL.RET_TIME]); lastCap = Number(row[COL.CAP]); lastTotal = Number(row[COL.TOTAL]); lastOutDepart = fmtTime(row[COL.OUT_DEPART] || ""); lastOutArrive = fmtTime(row[COL.OUT_ARRIVE] || ""); lastRetDepart = fmtTime(row[COL.RET_DEPART] || ""); lastRetArrive = fmtTime(row[COL.RET_ARRIVE] || ""); } if (lastPlate !== car) continue; if (!rowOrder || rowOrder === "－") continue; const paxObj = { orderNo: rowOrder, name: String(row[COL.NAME]).trim(), pax: Number(row[COL.PAX]), outStatus: String(row[COL.OUT_STATUS] || "").trim(), outTime: fmtTime(row[COL.OUT_TIME] || ""), retStatus: String(row[COL.RET_STATUS] || "").trim(), retTime: fmtTime(row[COL.RET_TIME2] || ""), }; if (lastSegment === "去程") { if (!depInfo) { depInfo = { vid: lastVid, plate: lastPlate, vtype: lastVtype, cap: lastCap, depPt: lastDepPt, depTime: lastDepTime, total: lastTotal, outDepart: lastOutDepart, outArrive: lastOutArrive, }; } depPassengers.push(paxObj); } else if (lastSegment === "回程") { if (!retInfo) { retInfo = { vid: lastVid, plate: lastPlate, vtype: lastVtype, cap: lastCap, retPt: lastDepPt, retTime: lastRetTime, total: lastTotal, retDepart: lastRetDepart, retArrive: lastRetArrive, }; } retPassengers.push(paxObj); } } const outConf = depPassengers.filter(p=>p.outStatus==="✅").reduce((s,p)=>s+p.pax,0); const retConf = retPassengers.filter(p=>p.retStatus==="✅").reduce((s,p)=>s+p.pax,0); const depTotal = depPassengers.reduce((s,p)=>s+p.pax,0); const retTotal = retPassengers.reduce((s,p)=>s+p.pax,0); return { depInfo, retInfo, depPassengers, retPassengers, summary: { depTotal, retTotal, outConfirmed: outConf, outPending: depTotal - outConf, retConfirmed: retConf, retPending: retTotal - retConf, } }; } // ============================================= // 取得總表 // ============================================= function getDashboard(date) { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const dateStr = fmtDate(date); const depMap = {}; const retMap = {}; let lastSegment = ""; let lastDate = ""; let lastPlate = ""; let lastVid = ""; let lastVtype = ""; let lastDepPt = ""; let lastDepTime = ""; let lastRetTime = ""; let lastOutDepart = ""; let lastOutArrive = ""; let lastRetDepart = ""; let lastRetArrive = ""; for (let i = 1; i < data.length; i++) { const row = data[i]; const rowSeg = String(row[COL.SEGMENT] || "").trim(); const rowDate = String(row[COL.DATE] || "").trim(); const rowPlate = String(row[COL.PLATE] || "").trim(); const rowOrder = String(row[COL.ORDER_NO]|| "").trim(); if (rowSeg) lastSegment = rowSeg; if (rowDate) lastDate = rowDate; if (!lastDate.includes(dateStr)) continue; if (rowPlate && rowPlate !== "？") { lastPlate = rowPlate; lastVid = String(row[COL.VID]).trim(); lastVtype = String(row[COL.VTYPE]).trim(); lastDepPt = String(row[COL.DEP_PT]).trim(); lastDepTime = fmtTime(row[COL.DEP_TIME]); lastRetTime = fmtTime(row[COL.RET_TIME]); lastOutDepart = fmtTime(row[COL.OUT_DEPART] || ""); lastOutArrive = fmtTime(row[COL.OUT_ARRIVE] || ""); lastRetDepart = fmtTime(row[COL.RET_DEPART] || ""); lastRetArrive = fmtTime(row[COL.RET_ARRIVE] || ""); } if (!lastPlate) continue; if (!rowOrder || rowOrder === "－") continue; const carKey = lastVid + "_" + lastDate; const map = lastSegment === "去程" ? depMap : retMap; } map[carKey].passengers.push({ orderNo: rowOrder, name: String(row[COL.NAME]).trim(), pax: Number(row[COL.PAX]), outStatus: String(row[COL.OUT_STATUS] || "").trim(), retStatus: String(row[COL.RET_STATUS] || "").trim(), }); } function summarizeCars(map) { return Object.values(map).map(car => { const total = car.passengers.reduce((s,p)=>s+p.pax, 0); const outConf = car.passengers.filter(p=>p.outStatus==="✅").reduce((s,p)=>s+p.pax, 0); const retConf = car.passengers.filter(p=>p.retStatus==="✅").reduce((s,p)=>s+p.pax, 0); return { ...car, totalPax:total, outConfirmed:outConf, retConfirmed:retConf, outPending:total-outConf, retPending:total-retConf }; }); } const depCars = summarizeCars(depMap); const retCars = summarizeCars(retMap); const depTotal = depCars.reduce((s,c)=>s+c.totalPax, 0); const retTotal = retCars.reduce((s,c)=>s+c.totalPax, 0); const depConf = depCars.reduce((s,c)=>s+c.outConfirmed, 0); const retConf = retCars.reduce((s,c)=>s+c.retConfirmed, 0); return { date: dateStr, depCars, retCars, summary: { depTotal, retTotal, depConfirmed: depConf, retConfirmed: retConf, depPending: depTotal-depConf, retPending: retTotal-retConf, } }; } // ============================================= // 更新乘客確認狀態 // ============================================= function updateStatus(orderNo, date, trip, status) { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const dateStr = fmtDate(date); const segment = trip === "outbound" ? } // ============================================= // 司機查詢：輸入車牌數字找車 // ============================================= function findCar(num, date) { if (!num) return { error: "請輸入車牌號碼" }; const sheet = getSheet(); const rowSeg = String(row[COL.SEGMENT] || "").trim(); const rowDate = String(row[COL.DATE] || "").trim(); const rowPlate = String(row[COL.PLATE] || "").trim(); const rowVid = String(row[COL.VID] || "").trim();
